# ALTERNATIVE si la chaîne dws-* n est pas disponible. Procédure recommandée : domino/PROCEDURE_TLONE.md
# ======================================================================
# TL-One — environnement Domino unique (à coller dans « Dockerfile Instructions »)
# Base conseillée : l'image Domino Standard Environment *GPU* (CUDA 12.x) de ta plateforme.
#
#   python système      -> l'APP Streamlit + Partie 2 (pas de modèle, CPU)
#   /opt/venv-tf5       -> Partie 1 DOM : Qwen3.8-27B-FP8   (transformers >= 5.9)
#   /opt/venv-tf4       -> v26 OV      : Qwen2.5-VL-7B      (transformers 4.x)
#
# Si pip passe par un miroir interne (Artifactory/Nexus), ajouter
#   ENV PIP_INDEX_URL=https://<miroir>/simple   PIP_TRUSTED_HOST=<miroir>
# Ajuster --index-url de torch à la version CUDA de l'image de base (cu121, cu124, cu126...).
# ======================================================================
USER root

# ---- 1. App + Partie 2 (python système de l'image)
RUN pip install --no-cache-dir \
      "streamlit>=1.40" "pandas>=2.0" "openpyxl>=3.1" "pymupdf>=1.23" "requests>=2.28" \
      "nbclient>=0.9" "nbformat>=5.9" "ipykernel>=6.0" "rapidfuzz"

# ---- 2. venv transformers 5 — Partie 1 (Qwen3.8-27B-FP8)
RUN python -m venv /opt/venv-tf5 && \
    /opt/venv-tf5/bin/pip install --no-cache-dir --upgrade pip && \
    /opt/venv-tf5/bin/pip install --no-cache-dir torch torchvision --index-url https://download.pytorch.org/whl/cu124 && \
    /opt/venv-tf5/bin/pip install --no-cache-dir \
      "transformers>=5.9" accelerate pymupdf pillow pandas psutil opencv-python-headless openpyxl \
      "nbclient>=0.9" "nbformat>=5.9" "ipykernel>=6.0"
# Optionnel (fast path Gated DeltaNet, cf. cellule 4 de la Partie 1) — compile, peut être long :
# RUN /opt/venv-tf5/bin/pip install --no-cache-dir causal-conv1d flash-linear-attention

# ---- 3. venv transformers 4 — v26 transferts (Qwen2.5-VL-7B, AutoModelForVision2Seq)
RUN python -m venv /opt/venv-tf4 && \
    /opt/venv-tf4/bin/pip install --no-cache-dir --upgrade pip && \
    /opt/venv-tf4/bin/pip install --no-cache-dir torch torchvision --index-url https://download.pytorch.org/whl/cu124 && \
    /opt/venv-tf4/bin/pip install --no-cache-dir \
      "transformers>=4.49,<5" accelerate qwen-vl-utils pymupdf pillow openpyxl numpy pandas \
      "nbclient>=0.9" "nbformat>=5.9" "ipykernel>=6.0"

RUN chmod -R a+rX /opt/venv-tf5 /opt/venv-tf4
USER ubuntu
