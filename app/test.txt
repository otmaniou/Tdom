import unicodedata
import pandas as pd

def normaliser_entete(valeur):
    texte = unicodedata.normalize("NFKC", str(valeur))
    return " ".join(texte.replace("\xa0", " ").split()).casefold()

def trouver_colonne(df, nom):
    cible = normaliser_entete(nom)

    for colonne in df.columns:
        if normaliser_entete(colonne) == cible:
            return colonne

    return None

col_date = trouver_colonne(df, "Request Decision Date")

if col_date is None:
    raise ValueError("Colonne Request Decision Date introuvable")

df["date_decision_normalisee"] = pd.to_datetime(
    df[col_date],
    dayfirst=True,
    errors="coerce"
)

print(
    "Dates de décision disponibles :",
    df["date_decision_normalisee"].notna().sum()
)
