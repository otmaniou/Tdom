import pandas as pd
import unicodedata

def normaliser_colonne(valeur):
    texte = unicodedata.normalize("NFKC", str(valeur))
    return " ".join(texte.replace("\xa0", " ").split()).casefold()

def trouver_colonne(df, nom):
    cible = normaliser_colonne(nom)
    for col in df.columns:
        if normaliser_colonne(col) == cible:
            return col
    return None

col_date = trouver_colonne(df, "Request Decision Date")

if col_date is not None:
    df["date_decision_normalisee"] = pd.to_datetime(
        df[col_date],
        dayfirst=True,
        errors="coerce"
    )
else:
    print("ALERTE : colonne Request Decision Date introuvable")
